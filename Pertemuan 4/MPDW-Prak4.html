generate_ma2_manual <- function(n, theta1, theta2) {
  burn <- 2
  e <- rnorm(n + burn)
  y <- numeric(n)
  for (t in 1:n) {
    tt <- t + burn
    y[t] <- e[tt] + theta1 * e[tt - 1] + theta2 * e[tt - 2]
  }
  y
}

scatter_lag <- function(y, lag, judul) {
  n <- length(y)
  plot(y[1:(n - lag)], y[(1 + lag):n],
       xlab = bquote(Y[t - .(lag)]), ylab = expression(Y[t]),
       main = judul, pch = 19, col = "steelblue")
}

Teoritis

persamaan Yule-Walker: \[\rho_1 = \frac{\phi_1}{1-\phi_2}, \qquad \rho_k = \phi_1 \rho_{k-1} + \phi_2 \rho_{k-2}\ \ (k \ge 2), \qquad \rho_0 = 1\]

phi1 <- 0.5
phi2 <- 0.2

rho <- numeric(3)
rho[1] <- phi1 / (1 - phi2)
rho[2] <- phi1 * rho[1] + phi2 * 1
rho[3] <- phi1 * rho[2] + phi2 * rho[1]

Teoritis

untuk lag \(k>q=2\), ACF ARMA(2,2) mengikuti persamaan selisih yang sama seperti AR(2):

\[\rho_k = \phi_1 \rho_{k-1} + \phi_2 \rho_{k-2}, \qquad k>2\]

sedangkan \(\rho_0,\rho_1,\rho_2\) dipengaruhi juga oleh komponen MA sehingga tidak memiliki bentuk tertutup sesederhana AR/MA murni — dalam praktik nilai-nilai ini dihitung memakai fungsi bawaan R, ARMAacf(), yang menghitungnya secara eksak melalui representasi \(MA(\infty)\).

acf_teori_arma <- ARMAacf(ar = c(0.5, 0.2), ma = c(0.4, 0.6), lag.max = 3)[2:4]
