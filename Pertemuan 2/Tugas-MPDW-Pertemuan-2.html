hildreth.lu.func<- function(r, model1){
  x <- model.matrix(model1)[,-1]
  y <- model.response(model.frame(model1))
  n <- length(y)
  t <- 2:n
  y <- y[t]-r*y[t-1]
  x <- x[t]-r*x[t-1]
  
  return(lm(y~x))
}

rOpt <- seq(0.2,0.5, by= 0.001)
